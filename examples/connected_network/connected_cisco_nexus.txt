hostname NEXUS-CORE-01
version 9.3

feature ospf
feature bgp
feature interface-vlan

vlan 10
  name DATA
vlan 20
  name VOICE

interface Ethernet1/1
  description Uplink to DIST-SW-01
  no switchport
  ip address 10.0.2.1/30
  no shutdown

interface Ethernet1/2
  description Connection to ASA Inside
  no switchport
  ip address 192.168.1.254/24
  no shutdown

interface Ethernet1/3
  description Connection to PA-FW Inside
  no switchport
  ip address 192.168.10.253/24
  no shutdown

interface loopback0
  description Router ID
  ip address 1.1.1.1/32

router ospf 1
  router-id 1.1.1.1
  network 10.0.2.0/30 area 0.0.0.0
  network 192.168.1.0/24 area 0.0.0.0
  network 192.168.10.0/24 area 0.0.0.0

router bgp 65001
  router-id 1.1.1.1
  neighbor 10.0.2.2 remote-as 65001
  neighbor 10.0.2.2 description DIST-SW-01
  address-family ipv4 unicast
    network 192.168.1.0/24
    network 192.168.10.0/24
