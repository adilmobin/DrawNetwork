hostname NEXUS-CORE-01
version 9.3

feature ospf
feature bgp
feature interface-vlan
feature vpc

vlan 10
  name DATA
vlan 20
  name VOICE
vlan 30
  name MGMT
vlan 100
  name SERVER

interface Ethernet1/1
  description Uplink to CORE-02
  no switchport
  ip address 10.0.1.1/30
  no shutdown

interface Ethernet1/2
  description Uplink to DIST-01
  no switchport
  ip address 10.0.2.1/30
  no shutdown

interface Ethernet1/3
  description Connection to Firewall
  switchport mode trunk
  switchport trunk allowed vlan 10,20,30,100
  no shutdown

interface Ethernet1/10
  description Server Port
  switchport access vlan 100
  no shutdown

interface Vlan10
  description Data VLAN
  ip address 192.168.10.1/24
  no shutdown

interface Vlan20
  description Voice VLAN
  ip address 192.168.20.1/24
  no shutdown

interface Vlan30
  description Management VLAN
  ip address 192.168.30.1/24
  no shutdown

interface loopback0
  description Router ID
  ip address 1.1.1.1/32

router ospf 1
  router-id 1.1.1.1
  network 10.0.1.0/30 area 0.0.0.0
  network 10.0.2.0/30 area 0.0.0.0
  network 192.168.10.0/24 area 0.0.0.0
  network 192.168.20.0/24 area 0.0.0.0

router bgp 65001
  router-id 1.1.1.1
  neighbor 10.0.1.2 remote-as 65001
  address-family ipv4 unicast
    network 192.168.10.0/24
    network 192.168.20.0/24

ip route 0.0.0.0/0 10.0.1.254
