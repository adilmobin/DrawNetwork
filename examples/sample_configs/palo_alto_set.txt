set network interface ethernet ethernet1/1 comment "Outside Interface"

set network interface ethernet ethernet1/2 comment "Inside Interface"
set network interface ethernet ethernet1/3 layer3 ip 172.16.10.1/24
set network interface ethernet ethernet1/3 comment "DMZ Interface"

set zone trust network layer3 ethernet1/2
set zone untrust network layer3 ethernet1/1
set zone dmz network layer3 ethernet1/3

set network virtual-router default interface ethernet1/1
set network virtual-router default interface ethernet1/2
set network virtual-router default interface ethernet1/3

set network virtual-router default protocol ospf enable yes

set network virtual-router default protocol ospf area 0.0.0.0 interface ethernet1/2 enable yes

set rulebase security rules allow-outbound from trust to untrust source any destination any application any service application-default action allow
set rulebase security rules allow-dmz-web from untrust to dmz source any destination dmz-servers application web-browsing service application-default action allow

set rulebase nat rules outbound-nat from trust to untrust source 192.168.1.0/24 destination any source-translation dynamic-ip-and-port interface-address interface ethernet1/1
set rulebase nat rules dmz-web-nat from untrust to dmz destination 203.0.113.25 source any destination-translation translated-address 172.16.10.10
